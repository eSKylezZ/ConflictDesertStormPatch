# src/generated/pse_ids_ds1.h, pse/include/pse/pse_ids_ds1.h and docs/sdk/reference/ids-ds1.md.
# Format + rules: docs/sdk/reference/ids.md. Ids listed in ds1.published.txt are frozen for ever: never removed,
# renumbered, renamed or given another kind. New ids take the next free number (never a freed one).
#
#   game <id> <title>
#   database <n>                                  (+1 by gen_ids.py --publish)
#   build <name> <pe timestamp> <image size> <sha256 | -> <notes>
#   [ids]                <id> <name> <kind> <description>      kind = function | site | global | vtable | data
#   [addresses <build>]  <id> <va> <signature>
#
# signature: hex bytes expected at <va> ("??" = any byte), or "@0x<va2>:<hex>" = bytes at another address (globals
# and IAT slots: a code window around an instruction that uses the address), "-" = not verified, "?" = to be filled
# by `python scripts/gen_ids.py --fill-signatures ds1` (from the exe). 4..16 bytes.
